བྱ་སྐྱེད་བྱེད་སོགས་ཀྱི་རྒྱུ་འབྲས་མི་རུང་བས་ཁྱབ་པར་སྨྲ་བ་ནི་དངོས་པོར་སྨྲ་བའི་ལུགས་ཡིན་པས། མགོན་པོ་ཀླུ་སྒྲུབ་ཀྱི་བཞེད་པ་རྒྱུ་དང་རྐྱེན་འདི་དང་འདི་ལ་བརྟེན་ནས་འབྲས་བུ་འདི་དང་འདི་སྐྱེ་བ་དང་འགག་པའི་རྒྱུ་འབྲས་ཀྱི་རྣམ་གཞག་ཉིད་ལ་བརྟེན་ནས་རང་བཞིན་གྱིས་སྟོང་པ་དང་དབུ་མའི་ལམ་བཙལ་དགོས་པ་ཡིན་ཏེ། རབ་བྱེད་ཉེར་བཞི་པ་ལས། རྟེན་ཅིང་འབྲེལ་པར་འབྱུང་བ་གང༌། །དེ་ནི་སྟོང་པ་ཉིད་དུ་བཤད། །དེ་ནི་བརྟེན་ནས་གདགས་པ་སྟེ། །དེ་ཉིད་དབུ་མའི་ལམ་ཡིན་ནོ། །གང་ཕྱིར་རྟེན་འབྱུང་མ་ཡིན་པའི། །ཆོས་འགའ་ཡོད་པ་མ་ཡིན་པ། །དེ་ཕྱིར་སྟོང་པ་མ་ཡིན་པའི། །ཆོས་འགའ་ཡོད་པ་མ་ཡིན་ནོ། །ཞེས་རྟེན་འབྱུང་ལ་རང་བཞིན་གྱིས་སྟོང་པས་ཁྱབ་པར་གསུངས་པ་འདི་ལ། རྒྱུ་རྐྱེན་ལ་བརྟེན་ནས་སྐྱེ་བ་ལ་རང་བཞིན་གྱིས་གྲུབ་པས་ཁྱབ་བོ་ཞེས་བཟློག་ནས་མ་སྨྲ་ཞིག །དེ་བཞིན་དུ་རྩོད་ཟློག་ལས་ཀྱང༌། གང་ལ་སྟོང་ཉིད་འདི་སྲིད་པ། །དེ་ལ་དོན་རྣམས་ཐམས་ཅད་སྲིད། །གང་ལ་སྟོང་ཉིད་མི་སྲིད་པ། །དེ་ལ་ཅི་ཡང་མི་སྲིད་དོ། །གང་ཞིག་སྟོང་དང་རྟེན་འབྱུང་དང༌། །དབུ་མའི་ལམ་དུ་དོན་གཅིག་པར། །གསུང་མཆོག་མཚུངས་པ་མེད་པ་ཡི། །སངས་རྒྱས་དེ་ལ་ཕྱག་འཚལ་ལོ། །ཞེས་དང༌། སྟོང་ཉིད་བདུན་ཅུ་པ་ལས་ཀྱང༌། དངོས་པོ་ཐམས་ཅད་རང་བཞིན་གྱིས། །སྟོང་པ་ཡིན་པས་དངོས་རྣམས་ཀྱི། །རྟེན་འབྱུང་ཕྱིར་ནི་དེ་བཞིན་གཤེགས། །མཚུངས་པ་མེད་པས་ཉེ་བར་བསྟན། །ཞེས་དང༌། རིགས་པ་དྲུག་ཅུ་པ་ལས་ཀྱང༌། གང་དག་གིས་ནི་མ་བརྟེན་པར། །བདག་གམ་འཇིག་རྟེན་མངོན་ཞེན་པ། །དེ་དག་ཀྱེ་མ་རྟག་མི་རྟག །ལ་སོགས་ལྟ་བས་འཕྲོག་པ་ཡིན། །གང་དག་བརྟེན་ནས་དངོས་པོ་རྣམས། །དེ་ཉིད་དུ་ནི་གྲུབ་འདོད་པ། །དེ་དག་

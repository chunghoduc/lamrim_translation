གང་ལ་བརྟེན་ནས་འབྱུང་ཡོད་པ། །དེ་ནི་རང་དབང་མི་འགྱུར་རོ། །འདི་ཀུན་རང་དབང་མེད་པ་སྟེ། །དེས་ན་བདག་ནི་ཡོད་མ་ཡིན། །ཞེས་གསུངས་ཤིང༌། དེའི་འགྲེལ་པ་ལས་ཀྱང༌། འདི་ན་གང་རང་གི་ངོ་བོ་དང་རང་བཞིན་དང་རང་དབང་དང་གཞན་ལ་རག་མ་ལས་པ་ཉིད་ཡིན་པ་དེ་ལ་ནི་རང་ལས་གྲུབ་པས་རྟེན་ཅིང་འབྲེལ་པར་འབྱུང་བ་ཡོད་པ་མ་ཡིན་ན། འདུས་བྱས་ཐམས་ཅད་རྟེན་ཅིང་འབྲེལ་པར་འབྱུང་བ་ཡང་ཡིན་ནོ། །

དེ་ལྟར་དངོས་པོ་གང་ལ་བརྟེན་ནས་འབྱུང་བ་ཡོད་པ་དེ་ནི་རང་དབང་དུ་མི་འགྱུར་ཏེ། རྒྱུ་དང་རྐྱེན་ལ་རག་ལས་ཏེ་སྐྱེ་བའི་ཕྱིར། འདི་ཀུན་རང་དབང་མེད་པ་སྟེ། དེའི་ཕྱིར་དངོས་པོ་གང་ལ་ཡང་བདག་སྟེ་རང་བཞིན་ཡོད་པ་མ་ཡིན་ནོ། །ཞེས་གསུངས་སོ། །རང་དབང་བ་ནི་རང་གི་ངོ་བོས་གྲུབ་པར་སྣང་བའི་ཚེ་ན་ཤེས་པ་དེ་དག་ལ་གཞན་ལ་རག་མ་ལས་སུ་སྣང་ཞིང༌། སྣང་བ་ལྟར་ཡང་གྲུབ་པའི་དོན་ནོ། །འོན་ཀྱང་རྒྱུ་རྐྱེན་གཞན་ལ་རག་མ་ལས་པ་ལ་བྱས་ནས་དེ་བཀག་ན་ནི་རང་སྡེ་ལ་བསྒྲུབ་མི་དགོས་པ་དང༌། དེ་ཁེགས་ཀྱང་དབུ་མའི་ལྟ་བ་རྙེད་པར་འཇོག་མི་ནུས་པས། ཡུལ་སྟེང་དུ་རང་གི་ངོ་བོའི་སྒོ་ནས་རང་ཚུགས་ཐུབ་པའི་སྡོད་ལུགས་ལ་རང་དབང་བར་བྱའོ། །དེས་ན་རང་བཞིན་གྱིས་སྟོང་པའི་དོན་ནི་རང་དབང་བའི་ངོ་བོ་དེ་དང་བྲལ་བ་ལ་བྱེད་ཀྱི། དོན་བྱེད་པའི་དངོས་པོ་མེད་པ་ལ་མི་བྱེད་པས། རྟེན་འབྲེལ་ཡིན་པའི་རྒྱུ་མཚན་གྱིས་རང་བཞིན་དགག་ནུས་པ་ཡིན་ཏེ། ལུང་སྔ་མའི་མཇུག་ཉིད་ནས། དེའི་ཕྱིར་འདིར་རྟེན་ཅིང་འབྲེལ་པར་འབྱུང་བ་རང་དབང་བའི་ངོ་བོ་དང་བྲལ་བའི་ཕྱིར། རང་དབང་བའི་ངོ་བོ་དང་བྲལ་བའི་དོན་སྟོང་པ་ཉིད་ཀྱི་དོན་ཡིན་གྱི།
